cmake_minimum_required(VERSION 3.18)
project(notflix VERSION 0.1.1 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(WEBOS "Build for an LG TV" OFF)
option(SANITIZE "Enable address and undefined behavior checks" OFF)
if(SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
find_package(PkgConfig REQUIRED)
foreach(lib IN ITEMS sdl2 SDL2_ttf SDL2_image libcurl)
  pkg_check_modules(${lib} REQUIRED IMPORTED_TARGET ${lib})
endforeach()
add_library(cjson STATIC third_party/cjson/cJSON.c)
target_include_directories(cjson PUBLIC third_party/cjson)
find_package(OpenSSL REQUIRED)
add_library(catalog src/catalog.c src/family.c src/captions.c)
target_link_libraries(catalog PUBLIC cjson PkgConfig::libcurl OpenSSL::Crypto)
target_include_directories(catalog PUBLIC src)
add_library(companion_http STATIC third_party/http/civetweb.c)
target_include_directories(companion_http PUBLIC third_party/http)
target_compile_definitions(companion_http PRIVATE NO_SSL NO_FILES NO_CGI NO_CACHING)
add_executable(notflix src/main.c src/view.c src/client.c src/screens.c src/layout.c src/account.c src/party.c src/companion.c src/bench.c third_party/qr/qrcodegen.c)
target_include_directories(notflix PRIVATE third_party/qr)
target_link_libraries(notflix PRIVATE companion_http dl pthread)
target_include_directories(notflix PRIVATE third_party/player)
target_link_libraries(notflix PRIVATE catalog PkgConfig::sdl2 PkgConfig::SDL2_ttf PkgConfig::SDL2_image m)
if(WEBOS)
  target_sources(notflix PRIVATE third_party/player/platform/webos/video_webos.c third_party/player/platform/webos/luna_webos.c)
  target_compile_definitions(notflix PRIVATE WEBOS SB_APP_ID="tv.notflix.native")
  target_link_libraries(notflix PRIVATE dl pthread)
  target_link_options(notflix PRIVATE "-Wl,--disable-new-dtags")
  set_target_properties(notflix PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/lib")
else()
  target_sources(notflix PRIVATE tests/desktop_video.c)
  enable_testing()
  add_executable(catalog-test tests/catalog.c)
  target_link_libraries(catalog-test PRIVATE catalog)
  add_test(NAME catalog COMMAND catalog-test)
  add_test(NAME navigation COMMAND notflix --self-test)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME addon-flow COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/addon.py" "$<TARGET_FILE:notflix>" "${CMAKE_SOURCE_DIR}/package")
  set_tests_properties(navigation PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}/package" ENVIRONMENT "SDL_VIDEODRIVER=dummy")
endif()
